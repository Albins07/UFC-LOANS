  <script>
  document.addEventListener('DOMContentLoaded', () => {
    const loanForm = document.getElementById('loanForm');
    const dateOutInput = document.getElementById('dateOut');
    const dateReturnInput = document.getElementById('dateReturn');
    const message = document.getElementById('message');
    const dateDisplay = document.getElementById('dateDisplay');

    const today = new Date().toISOString().split('T')[0];
    dateOutInput.min = today;

    dateOutInput.addEventListener('change', () => {
      dateReturnInput.min = dateOutInput.value;
      if (dateReturnInput.value < dateOutInput.value) dateReturnInput.value = '';
    });

    function showMessage(text, type = 'success') {
      message.textContent = text;
      message.className = `message ${type}`;
      message.classList.remove('hidden');
      setTimeout(() => message.classList.add('hidden'), 4000);
    }

    loanForm.addEventListener('submit', (e) => {
      e.preventDefault();

      const data = {
        item: document.getElementById('item').value,
        borrower: document.getElementById('borrower').value,
        dateOut: document.getElementById('dateOut').value,
        dateReturn: document.getElementById('dateReturn').value,
        notes: document.getElementById('notes').value
      };

      fetch('https://script.google.com/macros/s/AKfycbw2MfJ7dK7pS7BaHMkWlBuSViZt7K0izeo9Oehp5caiD57EJ4kLnyAqWhoH89rv7pnyMQ/exec', {
        method: 'POST',
        body: JSON.stringify(data),
        headers: { 'Content-Type': 'application/json' }
      })
      .then(res => res.json())
      .then(response => {
        if (response.status === 'success') {
          loanForm.reset();
          dateDisplay.classList.remove('hidden');
          document.getElementById('itemDisplay').textContent = data.item;
          document.getElementById('borrowerDisplay').textContent = data.borrower;
          document.getElementById('dateOutDisplay').textContent = data.dateOut;
          document.getElementById('dateReturnDisplay').textContent = data.dateReturn;
          showMessage('✅ Request submitted successfully!');
        } else {
          showMessage('Something went wrong!', 'error');
        }
      })
      .catch(err => {
        console.error('Error!', err.message);
        showMessage('Submission failed. Please try again.', 'error');
      });
    });
  });
  </script>
